<script>
  const STORAGE_KEY = 'signupUsers';

  const form = document.getElementById('signupForm');
  const usernameInput = document.getElementById('username');
  const emailInput = document.getElementById('email');
  const passwordInput = document.getElementById('password');
  const statusMsg = document.getElementById('statusMsg');
  const tableWrap = document.getElementById('tableWrap');

  function getUsers() {
    try {
      return JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];
    } catch (e) {
      return [];
    }
  }

  function saveUsers(users) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(users));
  }

  function clearErrors() {
    document.getElementById('usernameError').textContent = '';
    document.getElementById('emailError').textContent = '';
    document.getElementById('passwordError').textContent = '';
  }

  function validate(username, email, password, existingUsers) {
    let valid = true;

    if (!username.trim()) {
      document.getElementById('usernameError').textContent = 'Username is required.';
      valid = false;
    } else if (username.trim().length < 3) {
      document.getElementById('usernameError').textContent = 'Username must be at least 3 characters.';
      valid = false;
    } else if (existingUsers.some(u => u.username.toLowerCase() === username.trim().toLowerCase())) {
      document.getElementById('usernameError').textContent = 'That username is already taken.';
      valid = false;
    }

    const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!email.trim()) {
      document.getElementById('emailError').textContent = 'Email is required.';
      valid = false;
    } else if (!emailPattern.test(email.trim())) {
      document.getElementById('emailError').textContent = 'Enter a valid email address.';
      valid = false;
    } else if (existingUsers.some(u => u.email.toLowerCase() === email.trim().toLowerCase())) {
      document.getElementById('emailError').textContent = 'That email is already registered.';
      valid = false;
    }

    if (!password) {
      document.getElementById('passwordError').textContent = 'Password is required.';
      valid = false;
    } else if (password.length < 8) {
      document.getElementById('passwordError').textContent = 'Password must be at least 8 characters.';
      valid = false;
    } else if (!/[A-Z]/.test(password) || !/[0-9]/.test(password)) {
      document.getElementById('passwordError').textContent = 'Password needs at least 1 uppercase letter and 1 number.';
      valid = false;
    }

    return valid;
  }

  function renderTable() {
    const users = getUsers();

    if (users.length === 0) {
      tableWrap.innerHTML = '<div class="empty">No users yet. Sign up above to add one.</div>';
      return;
    }

    let html = '<table><thead><tr><th>Username</th><th>Email</th><th>Signed up</th><th></th></tr></thead><tbody>';
    users.forEach((u, i) => {
      html += `<tr>
        <td>${escapeHtml(u.username)}</td>
        <td>${escapeHtml(u.email)}</td>
        <td>${escapeHtml(u.createdAt)}</td>
        <td><button class="delete-btn" data-index="${i}">Delete</button></td>
      </tr>`;
    });
    html += '</tbody></table>';
    tableWrap.innerHTML = html;

    tableWrap.querySelectorAll('.delete-btn').forEach(btn => {
      btn.addEventListener('click', () => {
        const idx = parseInt(btn.getAttribute('data-index'), 10);
        const current = getUsers();
        current.splice(idx, 1);
        saveUsers(current);
        renderTable();
      });
    });
  }

  function escapeHtml(str) {
    const div = document.createElement('div');
    div.textContent = str;
    return div.innerHTML;
  }

  form.addEventListener('submit', (e) => {
    e.preventDefault();
    clearErrors();
    statusMsg.className = 'status';
    statusMsg.textContent = '';

    const username = usernameInput.value;
    const email = emailInput.value;
    const password = passwordInput.value;
    const users = getUsers();

    if (!validate(username, email, password, users)) {
      return;
    }

    users.push({
      username: username.trim(),
      email: email.trim(),
      password: password, // NOTE: plain-text storage — fine for a demo, never do this in a real app
      createdAt: new Date().toLocaleString()
    });

    saveUsers(users);
    form.reset();
    statusMsg.textContent = 'Account created successfully!';
    statusMsg.className = 'status success';
    renderTable();
  });

  renderTable();
</script>
